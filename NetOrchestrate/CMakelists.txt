CMAKE_MINIMUM_REQUIRED(VERSION 3.20)
PROJECT(API_Connector LANGUAGES CXX C)

SET(CMAKE_CXX_STANDARD 20)
SET(CMAKE_CXX_STANDARD_REQUIRED ON)

SET(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
SET(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
SET(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)


#setup boost library
find_package(OpenSSL REQUIRED)
find_package(Boost REQUIRED COMPONENTS headers json)
find_package(Boost REQUIRED COMPONENTS headers json thread chrono)
find_package(Threads REQUIRED)

add_library(API_Connector INTERFACE)


#include Memory Allocators : 
set(TOOLS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../GraphicsEngineSDK/Tools" CACHE PATH "Path to Tools directory")
cmake_path(SET TOOLS_DIR NORMALIZE "${TOOLS_DIR}")

message(STATUS "Tools directory: ${TOOLS_DIR}")
if(EXISTS "${TOOLS_DIR}")
    if (NOT TARGET LoggingSystem)
        add_library(LoggingSystem STATIC "${TOOLS_DIR}/LoggingSystem/src/LoggingSystem.cpp")

        target_include_directories(LoggingSystem PUBLIC "${TOOLS_DIR}/LoggingSystem/include")
    endif()
    
    if (NOT TARGET Allocators)
        add_library(Allocators INTERFACE)
        target_include_directories(Allocators INTERFACE "${TOOLS_DIR}/Allocators/include")
    endif()

else()
    message(FATAL_ERROR "Could not find Tools: ${TOOLS_DIR}")
endif()

if (NOT TARGET lob_engine)
# add lob_engine folder 
    set(LOB_ENGINE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../lob_engine" CACHE PATH "Path to lob_engine directory")
    if(EXISTS "${LOB_ENGINE_DIR}")
        add_subdirectory(${LOB_ENGINE_DIR} "${CMAKE_CURRENT_BINARY_DIR}/lob_engine")
    else()
        message(FATAL_ERROR "Could not find lob_engine: ${LOB_ENGINE_DIR}")
    endif()
endif()


target_include_directories(API_Connector INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
#add external includes 
target_include_directories(API_Connector INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/external/include)
target_link_libraries(API_Connector INTERFACE ${Boost_LIBRARIES})
target_link_libraries(API_Connector INTERFACE 
    Threads::Threads
    OpenSSL::SSL
    OpenSSL::Crypto
    Boost::headers
    Boost::json
    Boost::thread
    Boost::chrono
    Threads::Threads
)
#require the external tool libs 
target_link_libraries(
    API_Connector INTERFACE
    Allocators
)

add_executable(API_Connector_Tests
    tests/main.cpp
)

target_link_libraries(API_Connector_Tests PRIVATE API_Connector)
#link lob_engine
target_link_libraries(API_Connector_Tests PRIVATE
    lob_engine
)
